
#Importing data as artifact:
qiime tools import \
--type\
--input-path /path \
--input-format \
--output-path 

#trimming amplicon primers
qiime cutadapt trim-paired \
--i-XX.qza \
--p-cores 16 \
--p-\
--p-  \
--o-trimmed-sequences trimmed_sequence.qza \

#Denoising with DADA2 
qiime dada2 denoise-paired \
  --p-n-t \
  --i-XX.qza \
  --p-trim- \
  --p-trunc \
  --o-table table.qza \
  --o-XX.qza

#visualization
qiime feature-table summarize \
  --i-XX.qza \
  --o-XX.qzv \
  --m-sample-metadata-file 

qiime feature-table tabulate-seqs \
  --i-data .qza \
  --o- XX.qzv
qiime phylogeny align-to-tree-mafft-fasttree \
  --i-XX.qza \
  --o-XXseqs.qza \
  --o-masked-alignment masked-aligned-rep-seqs.qza \
  --o-tree.qza \
  --o-rooted-tree.qza

qiime diversity alpha-rarefaction \
  --i-table table.qza \
  --i-tree.qza \
  --p-max-depth \
  --m-metadata-file \
  --o-rarefaction.qzv


#Taxonomic Analysis
qiime feature-classifier \
  --i-classifier \
  --i-reads seqs.qza \
  --o-classification

qiime taxa barplot \
  --i-table \
  --i-taxonomy \
  --m-metadata-file\
  --o-visualization

  --input-path\
  --output-path
